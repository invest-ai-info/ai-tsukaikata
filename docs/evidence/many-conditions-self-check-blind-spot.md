### 照合コード（全版共通）

```python
import re

REQUIRED_WORDS = ["メンテナンス", "ログイン", "復旧"]
BANNED_WORDS = ["絶対", "保証"]
DATE_STR = "2026年9月10日"
DATE_BAD_PATTERNS = [r"9/10", r"9月10日\s*[（(]"]
TIME_GOOD = re.compile(r"\d{1,2}:\d{2}")
TIME_BAD_PATTERNS = [r"午前\d", r"午後\d", r"[０-９]時"]

def split_announcement(text):
    """区切り線「---」で分割し、箇条書き（・）をいちばん多く含むブロックを
    「お知らせ本文」とみなす（前置きの挨拶文・末尾の自己判定ブロックを除くため）。"""
    parts = re.split(r"\n-{3,}\n", text)
    parts = [p.strip() for p in parts if p.strip()]
    if not parts:
        return text.strip()
    return max(parts, key=lambda p: p.count("\n・"))

def lines_of(body):
    """先頭行が「#」見出し記法で、かつ次の行も短い見出し状の行
    （句点で終わらず、「・」でも始まらない）なら、先頭行は返答全体に
    付けた前置きのタイトルとみなして1行落とす（2つの見出し状の行が
    連続するのは、返答自体のタイトルとお知らせ本文の見出しが
    重複しているときだけなので、この形のときだけ外す）。"""
    ls = [l for l in body.split("\n") if l.strip() != ""]
    if ls and ls[0].lstrip().startswith("#") and len(ls) >= 2:
        nxt = ls[1].strip()
        looks_like_heading = (
            not nxt.endswith("。") and len(nxt) <= 40 and not nxt.startswith("・")
        )
        if looks_like_heading:
            ls = ls[1:]
    return ls

def bullet_lines(body):
    return [l for l in lines_of(body) if l.strip().startswith("・")]

def check_all(body):
    results = {}
    ls = lines_of(body)
    bullets = bullet_lines(body)
    heading = ls[0] if ls else ""

    results[1] = len(body.replace("\n", "")) <= 400
    results[2] = len(bullets) == 5
    results[3] = all(w in body for w in REQUIRED_WORDS)
    results[4] = all(w not in body for w in BANNED_WORDS)
    results[5] = DATE_STR in body and not any(re.search(p, body) for p in DATE_BAD_PATTERNS)
    results[6] = re.search(r"(である|だ。|だった。)", body) is None
    results[7] = "重要" not in heading
    last_line = ls[-1] if ls else ""
    results[8] = "問い合わせ" in last_line or "情報システム部" in last_line
    # 9. 見出しの字数は「実際に出力した1行目」をそのまま数える（記号込み）。
    #    自己申告がここを見落とすかどうかを見るのが、この記事の核。
    results[9] = len(heading.replace("・", "")) <= 15 if heading else False
    results[10] = all(len(b) <= 35 for b in bullets) if bullets else False
    if bullets:
        lead_lines = ls[1: ls.index(bullets[0])]
    else:
        lead_lines = ls[1:]
    results[11] = "".join(lead_lines).count("。") <= 2
    has_good_time = bool(TIME_GOOD.search(body))
    has_bad_time = any(re.search(p, body) for p in TIME_BAD_PATTERNS)
    results[12] = has_good_time and not has_bad_time
    return results
```

⚠️ **見出し行の特定について**: 一部の返りは、返答全体につけた「#」見出し（例：「# 社内向けお知らせ文」「# 経費精算システムメンテナンスのお知らせ」）のすぐ次の行に、お知らせ本文自身の見出し（「**システムメンテナンスのお知らせ**」「メンテナンスのお知らせ」）をもう一度書いていた。この「見出し状の行が2つ連続する」形のときだけ、1行目を前置きのタイトルとみなして落とし、2行目をお知らせ本文の見出しとして数えた（`lines_of` 関数）。この補正は、最初に n12版2回目を見たときに必要になって足したもので、その後 fix版1回目でも同じ形が出たため、同じ固定ルールをそのまま適用した（個々の回の結果を見て判定基準を変えてはいない）。「#」見出しの次が地の文（句点で終わる長い文）のときは補正の対象外で、その場合は「#」行をそのまま見出しとして数えている（n9版1回目がこのケース）。

## この記事から出た教訓（`content/_lessons.md` へ）

1. **4番が再発した**（AIの自己申告は、本人が自覚した範囲しか含まない）。今回の自覚できていない範囲は「自分がその場で付けた書式記号」。見出しの字数を条件にすると、AIは中身の文字数は正しく数えるが、自分で追加した `**`・`#` などの装飾記号は文字数に含めない。9回中4回、この理由で自己申告と実際がずれた。
