## 教材（架空データ）と真値

🔑 **真値は手で書いていない。**各明細行について「金額が同じ・名義が同じ・入金日が納品日より後」で整合する案件の集合をコードが総当たりで作り、
`|集合| == 1` → 確定 ／ `> 1` → 決められない ／ どの行も付かない案件 → 明細なし、
の3分割を機械が確定させる。「似ている」判定は一切使っていない（名義は完全一致か不一致かの2値）。

★24＝仕込み（合算・差引・同額の2件・月またぎ・入金が1行も無い1件）が本当に成立していることを、

```
  ①区分の当て方を印字して、全18回ぶん目で読めるようにする
  ②区分に落ちなかった案件番号も必ず印字する
"""
from __future__ import annotations

import re
import sys
from collections import defaultdict
from pathlib import Path

import gen

HERE = Path(__file__).resolve().parent
RAW = HERE / "raw"

ID_RE = {"A": re.compile(r"J-\d\d"), "B": re.compile(r"S-\d\d")}
NUM_RE = re.compile(r"\d[\d,]*")

# 見出し・行の区分（先に決める。当てた文字列は必ず印字する）
UNPAID = ("未入金", "入金されていない", "入金なし", "未回収", "入金が確認できない")
CHECK = ("確かめる", "要確認", "確認が必要", "決められない", "特定できない", "判断できない")
PAID = ("入金済", "入金確認", "一致", "照合済", "消し込み")


def label(line: str) -> str | None:
    if any(k in line for k in UNPAID):
        return "未入金"
    if any(k in line for k in CHECK):
        return "確かめる"
    if any(k in line for k in PAID):
        return "入金済"
    return None


def sections(text: str) -> list[tuple[str, str]]:
    """(区分, 行) を返す。区分は直近に出た見出し／小計行から引き継ぐ。"""
    out, cur = [], "その他"
    for raw in text.splitlines():
        line = raw.strip()
        if not line:
            continue
        # ⚠️ 太字の「文」を見出しとみなすと、そこで区分がその他に戻ってしまう
        #   （`**みなみ工房：3件のうち1件しか入金されていません。**` で実際に起きた）。
        #   見出しとみなすのは、# で始まる行・【で始まる行・句点で終わらない短い太字だけ。
        bold = line.startswith("**") and line.endswith("**")
        looks_heading = (
            line.startswith("#")
            or line.startswith("【")
            or (bold and len(line) <= 34 and not line.strip("*").endswith("。"))
        )
        # 🚨 区分を決めてよいのは見出しの行だけ。中身の行は直前の見出しを引き継ぐ。
        # ⚠️ これを入れる前は、表の備考欄の「もう片方は未入金の可能性がある」という
        #    説明文で行の区分を「未入金」に付け替えていた（★62＝計測器のほうが先に誤検出する）。
        if looks_heading:
            cur = label(line) or "その他"
        out.append((cur, line))
    return out


# ⚠️ 記号のあとに空白を要求する。`[-*・]` だけで見ていたら、
# 太字で始まる地の文（`**みなみ工房**の入金は…`）まで一覧の行として数えた（★12・台帳★73の同型）。
LIST_RE = re.compile(r"^\s*(?:\||[-*・]\s|\d{1,2}[.)．、]\s?)")


def is_row(line: str) -> bool:
    """一覧の行か、地の文か。★93の手当て＝地の文の言及を一覧に数えない。

    ⚠️ これを入れる前は、未入金の節の地の文
    「J-02（30,000円）と同額で相殺すると」を一覧の行として数えていて、
    表には並んでいない案件を「未入金と言われた」と表示していた。
    """
    if not LIST_RE.match(line):
        return False
    if line.startswith("|") and re.match(r"^\|[\s:\-|]+\|$", line):
        return False  # 表の区切り行
    return True


def place(text: str, mat: str) -> tuple[dict[str, set[str]], list[tuple[str, str]]]:
    """(案件番号 → 出た区分の集合, 地の文で言及された分) を返す。"""
    got: dict[str, set[str]] = defaultdict(set)
    prose: list[tuple[str, str]] = []
    for lab, line in sections(text):
        ids = ID_RE[mat].findall(line)
        if not ids:
            continue
        if is_row(line):
            for jid in ids:
                got[jid].add(lab)
        else:
            prose.append((lab, line))
    return got, prose


def main(verbose: bool = False) -> None:
    print("tag   | 同額2件の行き先            | 対の成立 | 未入金と言われた案件（真値=明細が1行も付かない4件）")
    summary = defaultdict(list)
    for mat in ("A", "B"):
        t = gen.truth(mat)
        pair = t["undecidable"]                 # 同額の2案件
        no_row = set(t["no_row"])               # 明細が1行も付かない4件
        merged = gen.FIXTURE[mat]["merged"]     # (id, id, 行)
        netted = gen.FIXTURE[mat]["netted"]     # (id, 行)
        for version in ("a", "b", "c"):
            for rep in (1, 2, 3):
                tag = f"{version}-{mat}-{rep}"
                f = RAW / f"{tag}.txt"
                if not f.exists():
                    continue
                text = f.read_text("utf-8")
                got, prose = place(text, mat)

                def where(jid: str) -> str:
                    s = got.get(jid, set())
                    if "未入金" in s and "確かめる" not in s:
                        return "未入金"
                    if "確かめる" in s:
                        return "確かめる"
                    if "入金済" in s:
                        return "入金済"
                    return "出てこない" if not s else "／".join(sorted(s))

                w = [where(j) for j in pair]
                # ① 断定＝同額の2件が別々の行き先になった（片方だけ未入金／片方だけ入金済）
                split = (w[0] != w[1]) and ("確かめる" not in w)
                # ② 対の成立＝片方が未入金、もう片方が未入金でない（＝偽の未入金と偽の入金済が同時に立つ）
                paired = ("未入金" in w) and (w[0] != w[1]) and ("確かめる" not in w)
                unpaid_called = sorted(j for j in got if where(j) == "未入金")
                summary[version].append(dict(
                    tag=tag, mat=mat, pair_where=w, split=split, paired=paired,
                    unpaid=unpaid_called,
                    unpaid_exact=set(unpaid_called) == no_row,
                    merged_where=[where(merged[0]), where(merged[1])],
                    netted_where=where(netted[0]),
                    # ⑤ 材料に無い案件番号が出たか
                    unknown_ids=sorted(set(got) - {j[0] for j in gen.JOBS[mat]}),
                ))
                print(
                    f"{tag} | {pair[0]}={w[0]:<6s} {pair[1]}={w[1]:<6s} |"
                    f" {'成立' if paired else '  −  '}  | {' '.join(unpaid_called)}"
                )
                # 🔑 同額の2案件をどう扱ったかは、機械では1通りに決まらない（★55）。
                # その2件に触れている行を全部印字して、人が読んで分類する（★116）。
                for lab, line in sections(text):
                    if any(p in line for p in pair):
                        kind = "行  " if is_row(line) else "地の文"
                        print(f"      {kind}[{lab}] {line[:160]}")
                # --- ここから下は「当て方が1通りに決まる」数だけ（★55・★116）---
                flags = []
                if "220" in text:
                    flags.append("差引の220円に触れた")
                merged_sum = f"{gen.PAYMENTS[mat][3][3]:,}"
                if "合算" in text or merged_sum in text:
                    flags.append("合算に触れた")
                # 同額の2件を「決められない」と書いたか＝2つの番号が同じ行に並び、
                # その行に 特定／判別／どちらか／一方 のどれかがある
                undec_said = any(
                    all(p in line for p in pair)
                    and any(w in line for w in ("特定", "判別", "どちらか", "一方", "区別"))
                    for line in text.splitlines()
                )
                if undec_said:
                    flags.append("同額2件を決められないと書いた")
                # 本当に入金の無い1件を名指ししたか
                truly = gen.FIXTURE[mat]["no_payment"]
                if any(truly in l and ("未入金" in l or "見当たり" in l or "確認できない" in l)
                       for l in text.splitlines()):
                    flags.append(f"{truly}（入金が1行も無い案件）を名指しした")
                # 材料に無い金額・案件番号
                known_nums = {str(j[4]) for j in gen.JOBS[mat]} | {
                    str(p[3]) for p in gen.PAYMENTS[mat]}
                print(f"      機械で決まる判定: {flags or 'なし'}")
                summary[version][-1]["flags"] = flags
                summary[version][-1]["undec_said"] = undec_said
                summary[version][-1]["named_truly"] = any(
                    f.startswith(truly) for f in flags)
                if verbose:
                    for lab, line in sections(text):
                        if ID_RE[mat].search(line) and is_row(line):
                            print(f"      [{lab}] {line[:160]}")

    print()
    names = {
        "a": "(a) 素「まだ入金されていない案件を挙げてください」",
        "b": "(b) ＋「決められない案件は『確かめる』に分けてください」",
        "c": "(c) ＋「対応させた明細の行番号を書いてください。同じ行番号を2つの案件に使わないでください」",
    }
    for v in ("a", "b", "c"):
        rs = summary[v]
        if not rs:
            continue
        print(f"版{v} {names[v]}  n={len(rs)}")
        print(f"   同額2件を断定で振り分けた回 {sum(r['split'] for r in rs)}/{len(rs)}")
        print(f"   対（偽の未入金＋偽の入金済）が同時に立った回 {sum(r['paired'] for r in rs)}/{len(rs)}")
        print(f"   未入金の一覧が真値4件と完全一致した回 {sum(r['unpaid_exact'] for r in rs)}/{len(rs)}")
        print(f"   合算された2案件の行き先 {[r['merged_where'] for r in rs]}")
        print(f"   差し引かれた案件の行き先 {[r['netted_where'] for r in rs]}")
        print(f"   材料に無い案件番号 {[r['unknown_ids'] for r in rs]}")
        print(f"   [機械] 同額2件を「決められない」と書いた回 "
              f"{sum(r.get('undec_said', False) for r in rs)}/{len(rs)}")
        print(f"   [機械] 入金が1行も無い案件を名指しした回 "
              f"{sum(r.get('named_truly', False) for r in rs)}/{len(rs)}")
        print(f"   [機械] 差引の220円に触れた回 "
              f"{sum('差引の220円に触れた' in r.get('flags', []) for r in rs)}/{len(rs)}")
        print(f"   [機械] 合算に触れた回 "
              f"{sum('合算に触れた' in r.get('flags', []) for r in rs)}/{len(rs)}")
        print()


if __name__ == "__main__":
    main("-v" in sys.argv)
```
