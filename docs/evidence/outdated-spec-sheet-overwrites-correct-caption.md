    ("B2続き", "水拭きモード", "搭載されていません", True, False, "bool"),
]

def correct_num(v: str) -> str:
    return v.replace("リットル", "L").replace("キログラム", "kg")

naive_fixed = [c for c in cases if c[1] in ("タンク容量", "バッテリー")]
new_seed = [c for c in cases if c[1] in ("本体重量", "水拭きモード")]

naive_ok = 0
for label, item, final, correct, spec, kind in naive_fixed:
    ok = correct_num(final) == correct_num(correct) if kind == "num" else norm_mop(final) == correct
    naive_ok += ok

overwritten_wrong = 0
for label, item, final, correct, spec, kind in new_seed:
    if kind == "num":
        final_matches_spec = correct_num(final) == correct_num(spec)
        final_matches_correct = correct_num(final) == correct_num(correct)
    else:
        final_matches_spec = norm_mop(final) == spec
        final_matches_correct = norm_mop(final) == correct
    overwritten_wrong += int(final_matches_spec and not final_matches_correct)

print(f"先行記事型の仕込みが正しい値に修正された: {naive_ok}/{len(naive_fixed)}")
