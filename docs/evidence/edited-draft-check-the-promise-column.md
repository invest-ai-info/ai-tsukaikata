    return NUM_RE.findall(norm(text).replace(",", "").replace("，", ""))

    t = norm(text)
    return [w for w in PROMISE_WORDS if w in t]

    t = norm(text)
    return [w for w in APOLOGY_WORDS if w in t]

    """(事実欄, 約束欄, 欄を見つけられたか) を返す。

    m = MATERIALS[mat]
    lines = text.splitlines()
    fi = pi = -1
    for i, line in enumerate(lines):
        if fi < 0 and m["fact_key"] in line and len(line) < 40:
            fi = i
        elif pi < 0 and m["promise_key"] in line and len(line) < 40:
            pi = i
    if fi < 0 or pi < 0 or pi <= fi:
        return text, "", False
    return "\n".join(lines[fi:pi]), "\n".join(lines[pi:]), True


if __name__ == "__main__":

def measure(text: str, mat: str) -> dict:

    for mat in ("A", "B"):
        src_fact_lines = [
            l.lstrip("・").strip()
            for l in gen.split_sections(gen.MATERIALS[mat]["text"], mat)[0].splitlines()
            if l.startswith("・")
        ]
        for rep in (1, 2):
            f = RAW / f"e-{mat}-{rep}.txt"
            if not f.exists():
                continue
            out = gen.norm(f.read_text("utf-8"))
            kept = [l for l in src_fact_lines if gen.norm(l) in out]
            print(
                f"  e-{mat}-{rep}: 事実欄の{len(src_fact_lines)}行のうち"
                f"一字一句そのまま残った行 {len(kept)}行"
                f"{'' if len(kept) == len(src_fact_lines) else ' ← 変わった行あり'}"
            )

    print()
    print("=== 版d の抜き出しが、原稿の行と一字一句同じか ===")
    for mat in ("A", "B"):
        src_lines = [
            l.lstrip("・").strip()
            for l in gen.MATERIALS[mat]["text"].splitlines()
            if l.startswith("・")
        ]
        for rep in (1, 2):
            f = RAW / f"d-{mat}-{rep}.txt"
            if not f.exists():
                continue
            out = f.read_text("utf-8")
            quoted = [
                l.lstrip("・-*").strip()
                for l in out.splitlines()
                if l.strip().startswith(("・", "- ", "* "))
            ]
            hit = [q for q in quoted if q in src_lines]
            miss = [q for q in quoted if q not in src_lines]
            print(
                f"  d-{mat}-{rep}: 引用として並んだ行 {len(quoted)}行／"
                f"原稿に一字一句あるもの {len(hit)}行／無いもの {len(miss)}行 {miss}"
            )

    print()
    print("=== 全回で出た約束語の内訳 ===")
    c = Counter()
    for v in ("a", "b", "c", "d", "e", "f"):
        for mat in ("A", "B"):
            for rep in range(1, reps[v] + 1):
                f = RAW / f"{v}-{mat}-{rep}.txt"
                if f.exists():
                    t = f.read_text("utf-8")
                    c.update(gen.promises(t))
                    c.update(gen.apologies(t))
    for w, n in c.most_common():
        print(f"  {w}: {n}回")


if __name__ == "__main__":
    main("-v" in sys.argv)
