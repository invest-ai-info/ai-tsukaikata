## 判定コード（judge.py）

```python
import re, glob

TRUE = {
    "a": {"sum": 67, "correct_avg": 67/11, "wrong_avg_14": 67/14,
          "zero_days": ["2026-08-05", "2026-08-10"],
          "fail_days": ["2026-08-03", "2026-08-07", "2026-08-12"]},
    "b": {"sum": 43, "correct_avg": 43/11, "wrong_avg_14": 43/14,
          "zero_days": ["2026-08-02", "2026-08-09"],
          "fail_days": ["2026-08-03", "2026-08-06", "2026-08-11"]},
}

def judge(material, text):
    t = TRUE[material]
    correct1, correct2 = f"{t['correct_avg']:.1f}", f"{t['correct_avg']:.2f}"
    result = {"correct_avg_found": (correct1 in text) or (correct2 in text)}
    result["denominator_14_as_final"] = False
    for m in re.finditer(r"(67|43)\s*(?:÷|/)\s*14", text):
        window = text[max(0, m.start() - 40):m.end() + 60]
        negated = any(w in window for w in
                      ["不適切", "低く", "誤って", "もし", "実態", "低く見積", "含めた場合", "参考", "低く出"])
        if not negated:
            result["denominator_14_as_final"] = True
    result["zero_days_named"] = all(d in text for d in t["zero_days"])
    conflated = []
    for d in t["fail_days"]:
        for m in re.finditer(re.escape(d), text):
            window = text[max(0, m.start() - 20):m.end() + 20]
            if "0件" in window and not any(
                w in window for w in ["できませんでした", "取得", "エラー", "失敗", "不明"]
            ):
                conflated.append(d)
    result["fail_day_conflated_with_zero"] = conflated
    return result

for path in sorted(glob.glob("out/v*_*.txt")):
    material = "a" if "_a_" in path else "b"
    text = open(path, encoding="utf-8").read()
    r = judge(material, text)
    print(path, r)
```

（＝取得失敗の3日を「0件」として書いた箇所が1件も無いこと）を確認してから記事に使っている
（台帳12番＝計測器のほうが間違っていることがあるため、コード自体も読み直した）。
