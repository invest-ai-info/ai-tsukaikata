## 指示文5（条件D'・埋もれた投稿文を、漠然と自己点検させる＝トラブル）

```python
MARKERS = ["PR", "ＰＲ", "広告", "プロモーション", "スポンサー"]

def classify(text: str):
    lines = text.split("\n")
    tag_line_idx = set()
    for i, line in enumerate(lines):
        tokens = line.split()
        hash_tokens = [t for t in tokens if t.startswith("#")]
        if len(hash_tokens) >= 3:
            tag_line_idx.add(i)
    has_disclosure = False
    first_line_hit = False
    in_tag_cluster = False
    matched_marker = None
    for i, line in enumerate(lines):
        for m in MARKERS:
            if m in line:
                has_disclosure = True
                matched_marker = m
                if i == 0:
                    first_line_hit = True
                if i in tag_line_idx:
                    in_tag_cluster = True
        if has_disclosure:
            break
    return {
        "has_disclosure": has_disclosure,
        "marker": matched_marker,
        "first_line": first_line_hit,
        "in_tag_cluster": in_tag_cluster,
    }

NG_WORDS = ["埋もれ", "気づけ", "移すべき", "分かりません", "認識しづらい"]
OK_WORDS = ["認識できます", "沿った表示", "問題ありません"]

def judge_verdict(text: str) -> str:
    has_ng = any(w in text for w in NG_WORDS)
    has_ok = any(w in text for w in OK_WORDS)
    if has_ng and not has_ok:
        return "NG"
    if has_ok and not has_ng:
        return "OK"
    return "unclear"

CATCH_WORDS = ["PR", "ＰＲ", "広告", "ハッシュタグ", "埋も", "気づけ", "分かりにく"]

def catches_disclosure_issue(text: str) -> bool:
    return any(w in text for w in CATCH_WORDS)
```
