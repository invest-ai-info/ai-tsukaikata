```python
import re, json

def classify_line(t):
    if re.search(r"まだ分からない|判定不能|判定不可|不明|わからない", t):
        return "unknown"
    if re.search(r"異常", t):
        return "stale"
    if re.search(r"正常", t):
        return "fresh"
    return "other"

def parse_response(text, ids):
    result = {}
    for _id in ids:
        m = re.search(re.escape(_id) + r"[^\n]{0,80}", text)
        result[_id] = classify_line(m.group(0)) if m else "MISSING"
    return result

# 指示文6（見出しで区分をまとめる回答形式）は上のシンプル版では拾えなかったため、
# セクション見出し（**正常**／**異常**／**まだ分からない**）を追跡する専用パーサを別途使った。
def parse_fix(text, ids):
    result, section = {}, None
    for raw in text.splitlines():
        line = raw.strip().lstrip("-").strip()
        if "**正常**" in raw: section = "fresh"
        elif "**異常**" in raw: section = "stale"
        elif "**まだ分からない**" in raw or "**不明**" in raw: section = "unknown"
        for i in ids:
            if line.startswith(i + ":") or line.startswith(i + " :"):
                inline = classify_line(line)
                result[i] = inline if inline else section
    return result
```
