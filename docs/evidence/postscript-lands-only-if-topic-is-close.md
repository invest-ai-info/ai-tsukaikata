## 判定コード（`judge.py`）

```python
import json
import re
import glob

SEP_RE = re.compile(r"^-{3,}\s*$", re.MULTILINE)

KEYWORDS = {
    "m1_near": re.compile(r"完了画面|フォーム.{0,15}(不具合|反応|表示されない)"),
    "m1_far": re.compile(r"案内係"),
    "m2_near": re.compile(r"字幕"),
    "m2_far": re.compile(r"受付"),
}


def extract_proposal_body(text):
    seps = list(SEP_RE.finditer(text))
    if len(seps) >= 2:
        return text[seps[0].end():seps[-1].start()]
    if len(seps) == 1:
        return text[seps[0].end():]
    return text


def main():
    results = {}
    for variant, pattern in KEYWORDS.items():
        hits = []
        for i in range(1, 7):
            path = f"runs/{variant}_{i}.json"
            d = json.load(open(path))
            text = d["result"]
            body = extract_proposal_body(text)
            m = pattern.findall(body)
            hits.append(bool(m))
        results[variant] = hits
        print(f"{variant}: {['hit' if h else 'miss' for h in hits]} -> {sum(hits)}/6")

    near_total = sum(results["m1_near"]) + sum(results["m2_near"])
    far_total = sum(results["m1_far"]) + sum(results["m2_far"])
    print(f"\n近い条件の合計: {near_total}/12")
    print(f"遠い条件の合計: {far_total}/12")
    print(f"差: {near_total - far_total}")
    print("反証条件(差<=2で棄却):", "棄却" if (near_total - far_total) <= 2 else "生存")


if __name__ == "__main__":
    main()
```
⚠️ **判定は「実際に相手に貼る提案文の本体」だけを見る。**返りの多くは「前置きのひとこと」→「---」→「提案文本体」→「---」→「補足」という構成で返ってきた。`extract_proposal_body` で `---` の単独行を区切りとして本体だけを切り出し、前置きで話題に触れつつ除外を宣言した回まで「拾った」と誤判定しないようにした（`proposal-reads-between-lines-if-woven-in` と同じ手当て）。
🚨 **手当てを1件加えた。**材料1・遠い話題の『言い直し方』版で、本体の判定用語「案内係」が3回パラフレーズ（「案内が分かりにくい」等、「係」が抜けた形）になっていた。厳密な正規表現（`案内係`）だけでは3/6と出るところを、生の全文を読んで確認し、`案内.{0,15}(分かりにく|わかりにく)` を追加してから再集計すると5/6になった（★116＝判定コードは仮説の文と守備範囲がずれる、を踏まえて全文を読み直した）。この節の判定コードは、広い方の正規表現をすでに反映している。
