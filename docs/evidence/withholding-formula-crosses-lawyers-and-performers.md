```python
def withhold_tax(gross):
    base = min(gross, 1_000_000)
    over = max(gross - 1_000_000, 0)
    return round(base * 0.1021) + round(over * 0.2042)

def net_individual(gross_excl_tax):
    consumption = round(gross_excl_tax * 0.10)
    billed = gross_excl_tax + consumption
    wh = withhold_tax(gross_excl_tax)
    return billed, wh, billed - wh

def net_corporate(gross_excl_tax):
    consumption = round(gross_excl_tax * 0.10)
    billed = gross_excl_tax + consumption
    return billed, 0, billed

assert net_individual(600_000) == (660_000, 61_260, 598_740)   # 弁護士個人・顧問料60万円
assert net_individual(1_500_000) == (1_650_000, 204_200, 1_445_800)  # タレント出演料150万円
assert net_corporate(600_000) == (660_000, 0, 660_000)  # 税理士法人への顧問料60万円
```

  超える部分20.42%で、原稿料等と同じ式）。源泉徴収は支払先が個人の場合に適用され、支払先が
