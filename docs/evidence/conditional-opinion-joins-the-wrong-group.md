## 機械照合コード（`check_merge.py`）

```python
import re
from pathlib import Path

COND = {
    "A": {"R09", "R10", "R11", "R12"},
    "B": {"S09", "S10", "S11", "S12"},
    "C": {"T01", "T02", "T09", "T10"},
}
ALL_IDS = {
    "A": {f"R{n:02d}" for n in range(1, 21)},
    "B": {f"S{n:02d}" for n in range(1, 23)},
    "C": {f"T{n:02d}" for n in range(1, 11)},
}
ID_RE = re.compile(r"\b([RST]\d{2})\b")


def group_ids(text: str) -> list[list[str]]:
    """テキストを「実際のグループ分けの単位」（見出しブロック・表の行）に割り、
    IDを含む単位だけを取り出す。「補足」以降の説明文は、実際のグループ分けでは
    ないので解析を打ち切る（初版はここが甘く、説明文中のID共起を誤って
    1グループと判定した＝台帳12番の再発）。
    """
    groups = []
    current_ids: list[str] = []
    for line in text.splitlines():
        if "補足" in line:
            break
        is_new_heading = bool(re.match(r"^\s*(#{1,6}\s|\*\*[①-⑳0-9]|##?#?\s*\d+\.|###?\s)", line))
        is_table_row = line.strip().startswith("|") and "|" in line.strip()[1:]
        ids_here = ID_RE.findall(line)
        if is_new_heading:
            if current_ids:
                groups.append(current_ids)
            current_ids = list(ids_here)
        elif is_table_row:
            if ids_here:
                groups.append(ids_here)
        else:
            if ids_here:
                current_ids.extend(ids_here)
    if current_ids:
        groups.append(current_ids)
    return [g for g in groups if g]


def analyze(material: str, path: Path) -> None:
    text = path.read_text(encoding="utf-8")
    cond = COND[material]
    groups = group_ids(text)
    misplaced = set()
    for g in groups:
        gset = set(g)
        if (gset & cond) and (gset - cond):
            misplaced |= (gset & cond)
    seen = set().union(*groups) if groups else set()
    missing = ALL_IDS[material] - seen
    print(f"{path.name}: misplaced_conditional={sorted(misplaced)} "
          f"({len(misplaced)}/{len(cond)}) | ids_not_seen_in_any_group={sorted(missing)}")
```

2. **新規（★157の隣）**：「条件付きの意見は別グループに」という一文単体で足りるとき、
   さらに「理由が違えば分けて」「件数を全部足して」を積み増しても、この実験では
   結果を1つも動かさなかった。素朴な指示文の失敗箇所（条件付きの意見だけ）を先に
   確認してから、そこだけを名指しする一文を足すほうが、効くかどうか分からない
   一文を積み増すより見通しが良かった。
