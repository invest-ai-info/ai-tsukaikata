# 試した証拠 — copy-ideas-second-round

**実測の相手**＝`claude --safe-mode --tools "" -p`（台帳★91の手順1）。
`--safe-mode` は CLAUDE.md・skills・plugins・hooks・過去の会話をすべて無効化するので、
呼び出しごとに独立した応答になる。同じ会話の2ターン目は `--session-id <uuid>` で
発行して `--resume` で続けた。作業ディレクトリはリポジトリ外の空フォルダ。
⚠️ ★20のとおり「事前知識ゼロ」とは言えないので、根拠にするのは機械照合できる結果だけにした。

**本体＝材料2本 × 5版 × 各2回 ＝ 20回**（1回につき一巡目と二巡目の2ターン）。
**後半の指示文＝材料2本 × 各2回**（f・g・h）。

        "latin_ok": LATIN_RE.search(idea) is None,
    }

    return all(judge(idea, mat).values())

    # ★24 = 仕込み（条件そのもの）が機械で1通りに決まることを、走らせる前に確かめる。
    # 通るはずの例と、条件ごとにちょうど1つだけ落ちる例を用意して assert する。

        assert passes(idea, mat) == want_pass, (mat, idea, got)
        if which is not None:
            failed = [k for k, v in got.items() if not v]
            assert failed == [which], (mat, idea, failed, "落ちる条件が1つに決まらない")
    # 必須語が禁止語と重ならないこと（重なると条件2と3が同時に満たせない）

# -*- coding: utf-8 -*-
"""返りから案を抜き出して、4条件・重複・言い換えの重なりを機械で数える。

台帳12番＝計測器のほうを先に疑う。だから
  ①抜き出しは2通り用意して、両方の数を印字する（★51）
  ②抜き出しに使わなかった行を全部印字する（落としたものが見える）

⚠️ 1回目に書いた抜き出しは「行頭の記号」しか見ておらず、
   記号を付けずに30行返した回（b-A-1）を **0案** と表示した。
   これを直したのがこのファイル。1回目の数字で書いていたら
   「『あと30案』と頼むと1案も返らない回がある」という嘘になっていた。
"""
from __future__ import annotations

import re
import unicodedata
from collections import defaultdict
from pathlib import Path

import gen

HERE = Path(__file__).resolve().parent
RAW = HERE / "raw"

MARK_RE = re.compile(r"^\s*(?:[・\-*＊•]|\d{1,2}[.)．、]|（\d{1,2}）)\s*")
DECLARE_RE = re.compile(r"(\d{1,3})\s*案")
MAX_IDEA_LEN = 30


def _strip(line: str) -> str:
    return MARK_RE.sub("", line.strip()).strip().strip("「」『』")


def extract_mark(text: str) -> list[str]:
    """当て方1＝行頭に記号か番号がある行だけ。無ければ当て方2に倒す。"""
    got = [_strip(l) for l in text.splitlines() if MARK_RE.match(l.strip())]
    got = [g for g in got if g]
    return got if len(got) >= 5 else extract_short(text)


def extract_short(text: str) -> list[str]:
    """当て方2＝記号を見ず、「30字以内で句点で終わらない行」を案とみなす。"""
    out = []
    for line in text.splitlines():
        s = _strip(line)
        if not s or s.endswith("。") or len(s) > MAX_IDEA_LEN:
            continue
        out.append(s)
    return out


def dropped_lines(text: str, ideas: list[str]) -> list[str]:
    kept = set(ideas)
    return [
        _strip(l) for l in text.splitlines()
        if l.strip() and _strip(l) not in kept
    ]


def norm(s: str) -> str:
    s = unicodedata.normalize("NFKC", s)
    return re.sub(r"[\s、。！？!?,.・「」『』（）()／/…—ー〜~]", "", s)


def bigrams(s: str) -> set[str]:
    s = norm(s)
    return {s[i:i + 2] for i in range(len(s) - 1)}


def near(a: str, b: str) -> float:
    """2案の文字2つ組の重なり（Dice係数）。言い換えだけの水増しを見るため。"""
    x, y = bigrams(a), bigrams(b)
    if not x or not y:
        return 0.0
    return 2 * len(x & y) / (len(x) + len(y))


def main() -> None:
    rows = []
    all_dropped = []
    for mat in ("A", "B"):
        for version in ("a", "b", "c", "d", "e"):
            for rep in (1, 2):
                tag = f"{version}-{mat}-{rep}"
                f1, f2 = RAW / f"{tag}-t1.txt", RAW / f"{tag}-t2.txt"
                if not f2.exists():
                    continue
                t1, t2 = f1.read_text("utf-8"), f2.read_text("utf-8")
                i1a, i2a = extract_mark(t1), extract_mark(t2)
                i1b, i2b = extract_short(t1), extract_short(t2)
                all_dropped += [(tag + "-t1", x) for x in dropped_lines(t1, i1a)]
                all_dropped += [(tag + "-t2", x) for x in dropped_lines(t2, i2a)]

                declared = None
                for line in dropped_lines(t2, i2a):
                    m = DECLARE_RE.search(line)
                    if m:
                        declared = int(m.group(1))
                        break

                exact = len(set(i1a) & set(i2a))
                loose = len({norm(x) for x in i1a} & {norm(x) for x in i2a})
                # 言い換えだけの重なり＝1巡目のどれかと Dice 0.7 以上
                nearly = sum(1 for y in i2a if any(near(x, y) >= 0.7 for x in i1a))

                rows.append(dict(
                    tag=tag, mat=mat, version=version, rep=rep,
                    n1=len(i1a), n2=len(i2a), n1b=len(i1b), n2b=len(i2b),
                    declared=declared,
                    pass1=sum(1 for x in i1a if gen.passes(x, mat)),
                    pass2=sum(1 for x in i2a if gen.passes(x, mat)),
                    exact=exact, loose=loose, nearly=nearly,
                    dup2=len(i2a) - len(set(i2a)),
                ))

    print("tag     | 1巡目 2巡目(当て方2) 申告 | 通過1 通過2 | 完全一致 正規化一致 似た案 | 2巡目内の重複")
    for r in rows:
        print(
            f"{r['tag']:8s}| {r['n1']:3d}/{r['n1b']:<3d} {r['n2']:3d}/{r['n2b']:<3d}"
            f" {str(r['declared']):>5s} | {r['pass1']:5d} {r['pass2']:5d} |"
            f" {r['exact']:8d} {r['loose']:10d} {r['nearly']:6d} | {r['dup2']:3d}"
        )

    print()
    print("=== 版ごとのまとめ（材料2本×各2回＝4回ぶん）===")
    by = defaultdict(list)
    for r in rows:
        by[r["version"]].append(r)
    names = {
        "a": "「もっと案をください。」",
        "b": "「あと30案出してください。」",
        "c": "「いま出た30案と一つも重ならない案を、30案出してください。」",
        "d": "1巡目とまったく同じ指示文を、同じ会話でもう一度",
        "e": "1巡目とまったく同じ指示文を、新しい会話でもう一度（対照）",
    }
    for v in ("a", "b", "c", "d", "e"):
        rs = by[v]
        if not rs:
            continue
        print(
            f"版{v} {names[v]}\n"
            f"   2巡目に返った案 {[r['n2'] for r in rs]}"
            f"（当て方2では {[r['n2b'] for r in rs]}）\n"
            f"   4条件を通った案 {[r['pass2'] for r in rs]}\n"
            f"   1巡目と完全一致 {[r['exact'] for r in rs]}"
            f" / 記号を落として一致 {[r['loose'] for r in rs]}\n"
            f"   1巡目の案と文字が7割以上重なる案 {[r['nearly'] for r in rs]}\n"
            f"   2巡目の中の重複 {[r['dup2'] for r in rs]}"
        )

    print()
    print("=== 1巡目の案と文字が7割以上重なった組（全部そのまま印字する）===")
    for mat in ("A", "B"):
        for version in ("a", "b", "c", "d", "e"):
            for rep in (1, 2):
                tag = f"{version}-{mat}-{rep}"
                f1, f2 = RAW / f"{tag}-t1.txt", RAW / f"{tag}-t2.txt"
                if not f2.exists():
                    continue
                i1 = extract_mark(f1.read_text("utf-8"))
                i2 = extract_mark(f2.read_text("utf-8"))
                for y in i2:
                    best = max(i1, key=lambda x: near(x, y))
                    s = near(best, y)
                    if s >= 0.7:
                        print(f"  [{tag}] {s:.2f}  1巡目「{best}」／2巡目「{y}」")

    print()
    print(f"=== 抜き出しに使わなかった行（全{len(all_dropped)}行・目で読む）===")
    for tag, line in all_dropped:
        print(f"  [{tag}] {line}")


if __name__ == "__main__":
    main()

   返りの上のほうだけをコピーすると、そこは丸ごと落ちる。

4. **既存記事の助言を測ったら、いちばん重なる形だった（新しい観察）。**
   `count-copy-ideas-that-pass` は「もう一度そのまま通してください」（＝新しい会話でやり直す）を勧めていたが、
   一巡目と文字が七割以上重なる案は、同じ会話で追加した12回が 0〜3件なのに対し、
