```python
# -*- coding: utf-8 -*-
"""架空の売上表を2本作り、真値を先に確定する（★24＝仕込みのほうも機械で確かめる）。

材料A＝きれいな表（日付 / 担当 / 商品 / 数量 / 単価 / 金額）30行。
材料B＝列の形を変えた表（受注No / 計上日 / 商品名 / 担当者 / 単価 / 数量 / 金額）30行。
        空欄（金額が入っていない行）と表記ゆれ（担当者名の全角空白あり／なし）を仕込む。

真値は「担当ごとの売上合計」。31行目を足したあとの真値も先に出しておく。
"""
from __future__ import annotations

import json
from pathlib import Path

HERE = Path(__file__).resolve().parent

# ---------------------------------------------------------------- 材料A
A_HEAD = ["日付", "担当", "商品", "数量", "単価", "金額"]
A_STAFF = ["山田", "佐藤", "鈴木"]
A_ITEMS = [
    ("A-100 定期点検パック", 12000),
    ("A-200 出張設置", 18000),
    ("A-300 消耗品セット", 3500),
    ("A-400 延長保証", 8000),
    ("A-500 初期設定代行", 15000),
]

A_ROWS = []
for i in range(30):
    day = 1 + (i * 13) % 28
    staff = A_STAFF[i % 3]
    name, unit = A_ITEMS[(i * 7) % 5]
    qty = 1 + (i * 5) % 4
    A_ROWS.append(
        {
            "日付": f"2026-07-{day:02d}",
            "担当": staff,
            "商品": name,
            "数量": qty,
            "単価": unit,
            "金額": qty * unit,
        }
    )

# 31行目（8月に足される1行）
A_EXTRA = {
    "日付": "2026-08-03",
    "担当": "佐藤",
    "商品": "A-200 出張設置",
    "数量": 2,
    "単価": 18000,
    "金額": 36000,
}

# ---------------------------------------------------------------- 材料B
B_HEAD = ["受注No", "計上日", "商品名", "担当者", "単価", "数量", "金額"]
B_STAFF = ["高橋 みどり", "高橋みどり", "中村 健", "中村健"]  # 表記ゆれを仕込む
B_ITEMS = [
    ("バナー制作 (小)", 5000),
    ("バナー制作 (大)", 9000),
    ("記事執筆 2000字", 6000),
    ("記事執筆 4000字", 11000),
    ("写真レタッチ 10点", 7000),
]

B_ROWS = []
for i in range(30):
    day = 2 + (i * 11) % 27
    # 表記ゆれ: 同じ人が、行によって空白ありと空白なしで書かれている
    if i % 2 == 0:
        staff = B_STAFF[0] if (i // 2) % 2 == 0 else B_STAFF[1]
    else:
        staff = B_STAFF[2] if (i // 2) % 2 == 0 else B_STAFF[3]
    name, unit = B_ITEMS[(i * 3) % 5]
    qty = 1 + (i * 4) % 3
    row = {
        "受注No": f"R-{2601 + i}",
        "計上日": f"2026/7/{day}",
        "商品名": name,
        "担当者": staff,
        "単価": unit,
        "数量": qty,
        "金額": qty * unit,
    }
    B_ROWS.append(row)

# 空欄の仕込み: 3行の「金額」を空欄にする（単価と数量はある＝計算はできる）
for idx in (4, 15, 26):
    B_ROWS[idx]["金額"] = ""

B_EXTRA = {
    "受注No": "R-2631",
    "計上日": "2026/8/1",
    "商品名": "記事執筆 4000字",
    "担当者": "中村健",
    "単価": 11000,
    "数量": 2,
    "金額": 22000,
}


def _norm(name: str) -> str:
    return name.replace(" ", "").replace("　", "")


def totals_a(rows):
    out = {}
    for r in rows:
        out[r["担当"]] = out.get(r["担当"], 0) + r["金額"]
    return out


def totals_b(rows, use_blank_as_zero: bool):
    """金額欄が空欄の行の扱いを2通りで出す。

    use_blank_as_zero=True  ＝ 金額欄をそのまま足す（空欄は0）
    use_blank_as_zero=False ＝ 単価×数量で埋めてから足す
    """
    out = {}
    for r in rows:
        key = _norm(r["担当者"])
        if r["金額"] == "":
            v = 0 if use_blank_as_zero else r["単価"] * r["数量"]
        else:
            v = r["金額"]
        out[key] = out.get(key, 0) + v
    return out


def totals_b_raw(rows, use_blank_as_zero: bool):
    """表記ゆれをまとめずに、書いてある文字列そのままで集計する。"""
    out = {}
    for r in rows:
        key = r["担当者"]
        if r["金額"] == "":
            v = 0 if use_blank_as_zero else r["単価"] * r["数量"]
        else:
            v = r["金額"]
        out[key] = out.get(key, 0) + v
    return out


def render(head, rows):
    lines = ["\t".join(head)]
    for r in rows:
        lines.append("\t".join(str(r[h]) for h in head))
    return "\n".join(lines)


def render_numbered(head, rows, start=2):
    """行番号つき（表計算ソフトに貼ったときの行番号）。"""
    lines = [f"1\t" + "\t".join(head)]
    for i, r in enumerate(rows):
        lines.append(f"{start + i}\t" + "\t".join(str(r[h]) for h in head))
    return "\n".join(lines)


if __name__ == "__main__":
    (HERE / "material_a.txt").write_text(render(A_HEAD, A_ROWS), encoding="utf-8")
    (HERE / "material_b.txt").write_text(render(B_HEAD, B_ROWS), encoding="utf-8")
    (HERE / "material_a_numbered.txt").write_text(
        render_numbered(A_HEAD, A_ROWS), encoding="utf-8"
    )
    (HERE / "material_b_numbered.txt").write_text(
        render_numbered(B_HEAD, B_ROWS), encoding="utf-8"
    )

    truth = {
        "A": {
            "30行": totals_a(A_ROWS),
            "31行(1行足したあと)": totals_a(A_ROWS + [A_EXTRA]),
            "足した行": A_EXTRA,
            "行の範囲": "2行目〜31行目（見出しが1行目）",
        },
        "B": {
            "30行_空欄は0_表記ゆれをまとめる": totals_b(B_ROWS, True),
            "30行_空欄は単価x数量_表記ゆれをまとめる": totals_b(B_ROWS, False),
            "30行_空欄は0_文字列そのまま": totals_b_raw(B_ROWS, True),
            "31行_空欄は0_表記ゆれをまとめる": totals_b(B_ROWS + [B_EXTRA], True),
            "31行_空欄は0_文字列そのまま": totals_b_raw(B_ROWS + [B_EXTRA], True),
            "足した行": B_EXTRA,
            "空欄の行": [i + 2 for i, r in enumerate(B_ROWS) if r["金額"] == ""],
            "行の範囲": "2行目〜31行目（見出しが1行目）",
        },
    }
    (HERE / "truth.json").write_text(
        json.dumps(truth, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(json.dumps(truth, ensure_ascii=False, indent=2))
```

### 返ってきたもの（生）— A-P1-1

```
担当ごとの売上合計は SUMIF で出せます。

■ 前提

貼っていただいた表が、Excel のシートの A1 から始まっているものとします。
1行目が見出し、2行目〜31行目がデータです。

  A列 日付 / B列 担当 / C列 商品 / D列 数量 / E列 単価 / F列 金額

■ 集計表を作る

空いている列（たとえば H列・I列）に、担当名を書き出します。

  H1: 担当   I1: 売上合計
  H2: 山田
  H3: 佐藤
  H4: 鈴木

■ I2 に入れる数式

  =SUMIF($B$2:$B$31, H2, $F$2:$F$31)

I2 を I4 まで下にコピーしてください。

  ・$B$2:$B$31 … 担当の列（探す範囲）
  ・H2         … 探す担当名
  ・$F$2:$F$31 … 金額の列（足す範囲）

■ 結果

  山田  302,500
  佐藤  269,500
  鈴木  265,500

■ 検算

  =SUM($F$2:$F$31)  →  837,500

3人の合計 837,500 と一致すれば、担当名の書き間違いや抜けはありません。
```

```

■ 数式

  I1: 担当者   J1: 売上合計
```

```python
FORMULA_RE = re.compile(r"=(?:SUMIF|SUMIFS|SUM|UNIQUE|FILTER)\b[^\n]*")

# 参照範囲の形
RANGE_FIXED = re.compile(r"\$?[A-Z]\$?\d+:\$?[A-Z]\$?(\d+)")   # $B$2:$B$31
RANGE_WHOLE = re.compile(r"\$?([A-Z]):\$?\1(?![0-9])")          # $B:$B
RANGE_TABLE = re.compile(r"テーブル\d*\[")                       # テーブル1[担当]


def classify(formula: str) -> str:
    if RANGE_TABLE.search(formula):
        return "テーブル参照"
    if RANGE_WHOLE.search(formula):
        return "列ぜんぶ"
    m = RANGE_FIXED.search(formula)
    if m:
        end = int(m.group(1))
        if end <= 31:
            return "最終行で固定"
        return "余裕を持たせた固定"
    return "その他"


def split_runs(text: str):
    """`## ラベル` ごとに、「返ってきたもの（生）」のコードブロックだけを取り出す。"""
    out = {}
    blocks = re.split(r"\n## ", text)
    for b in blocks[1:]:
        label = b.splitlines()[0].strip()
        m = re.search(r"### 返ってきたもの（生）\n\n```\n(.*?)\n```", b, re.S)
        out[label] = m.group(1) if m else ""
    return out


# ------------------------------------------------------------------ SUMIF を実際に計算する
def col_values(rows, head, col_letter):
    idx = ord(col_letter) - ord("A")
    return [r[head[idx]] for r in rows]


def sumif(cond_vals, criteria, sum_vals):
    total = 0
    wildcard = criteria.endswith("*")
    key = criteria[:-1] if wildcard else criteria
    for c, s in zip(cond_vals, sum_vals):
        c = str(c)
        hit = c.startswith(key) if wildcard else (c == criteria)
        if hit and isinstance(s, int):
            total += s
    return total

    """条件列と合計列は材料ごとに決まっている（A＝B列とF列／B＝D列とG列）。"""
    cond_col, sum_col = ("B", "F") if material == "A" else ("D", "G")

    def run(rows):
        cv = col_values(rows, head, cond_col)
        sv = col_values(rows, head, sum_col)
        if end_row is not None:
            n = end_row - 1  # 2行目〜end_row行目 ＝ n件
            cv, sv = cv[:n], sv[:n]
        return sumif(cv, criteria, sv)

    return run(rows30), run(rows31)
```
